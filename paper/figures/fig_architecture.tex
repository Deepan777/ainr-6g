\begin{tikzpicture}[

  blk/.style={draw, line width=0.6pt, rounded corners=1.5pt, align=center, inner sep=3pt,
              minimum height=9.5mm, minimum width=15mm, fill=white},
  trn/.style={blk, fill=gray!18},
  det/.style={->, line width=1.1pt},
  mon/.style={->, line width=0.6pt, draw=black!75},
  ada/.style={->, line width=0.8pt, dash pattern=on 4pt off 2pt},
  lab/.style={font=\footnotesize, inner sep=1.5pt},
]
% ------------------------------------------------------------------ detection path
\node[blk]  (y)   at (0,0)       {$\Y$};
\node[blk]  (ls)  at (0,-1.45)   {$\hat{\mathbf{H}}_{\mathrm{LS}}$};
\node[trn, minimum height=24mm, minimum width=16mm] (enc) at (2.55,-0.72) {$f_\theta$};
\node[trn]  (qc)  at (5.45,0)    {$q_\theta(\bc)$};

\draw[det] (ls) -- (ls -| enc.west);
\draw[line width=0.6pt, ->] (y.south) -- (ls.north);
\draw[det] (enc.east |- qc) -- (qc);

\node[blk]  (g)   at (8.35,-2.55) {$\tilde H(\bh)\,X(\bc)$};

\begin{scope}[on background layer]
  \node[fill=gray!7, draw=gray!45, line width=0.5pt, rounded corners=3pt, inner xsep=3mm, inner ysep=2.8mm,

\end{scope}
\node[lab, anchor=south west, text=black!70] at (panel.north west) {Physics-based self-assessment};
